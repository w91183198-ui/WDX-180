var vs let vs const
var let const
stored in global context ✅ ❌ ❌ 
                  
// 🟩 1. STORED IN GLOBAL CONTEXT
console.log("=== 1. Stored in Global Context ===");
                  
var globalVar = "I'm global with var";  // ✅ attached to window/globalThis
let globalLet = "I'm not global with let";  // ❌ not attached
const globalConst = "I'm not global with const"; // ❌ not attached
                  
console.log("window.globalVar:", window.globalVar); // ✅ visible
console.log("window.globalLet:", window.globalLet); // ❌ undefined
console.log("window.globalConst:", window.globalConst); // ❌ undefined
console.log("");
function scope ✅ ✅ ✅ 
                  
// 🟩 2. FUNCTION SCOPE
console.log("=== 2. Function Scope ===");
                  
function testScope() {
  var a = "var inside function";
  let b = "let inside function";
  const c = "const inside function";
  console.log("Inside function:", a, b, c);
}
testScope();
                  
try { console.log(a); } catch { console.log("a not accessible outside function ✅"); }
try { console.log(b); } catch { console.log("b not accessible outside function ✅"); }
try { console.log(c); } catch { console.log("c not accessible outside function ✅"); }
console.log("");
block scope ❌ ✅ ✅ 
                  
// 🟩 3. BLOCK SCOPE
console.log("=== 3. Block Scope ===");
                  
if (true) {
  var x = "var block";
  let y = "let block";
  const z = "const block";
  console.log("Inside block:", x, y, z);
}
console.log("Outside block var:", x); // ✅ var leaks out
try { console.log(y); } catch { console.log("let not accessible outside block ✅"); }
try { console.log(z); } catch { console.log("const not accessible outside block ✅"); }
console.log("");
can be reassigned? ✅ ✅ ❌ 
                  
// 🟩 4. CAN BE REASSIGNED
console.log("=== 4. Can Be Reassigned ===");
                  
var countVar = 1;
let countLet = 1;
const countConst = 1;
                  
countVar = 2; // ✅
countLet = 2; // ✅
try { countConst = 2; } catch { console.log("const cannot be reassigned ❌"); }
                  
console.log("var:", countVar, "| let:", countLet, "| const:", countConst);
console.log("");
can be redeclared? ✅ ❌ ❌ 
                  
// 🟩 5. CAN BE REDECLARED
console.log("=== 5. Can Be Redeclared ===");
                  
var redeclareVar = "first";
var redeclareVar = "second"; // ✅ allowed
console.log("var redeclared:", redeclareVar);
                  
try {
  let redeclareLet = "first";
  let redeclareLet = "second"; // ❌
} catch (err) {
  console.log("let cannot be redeclared ❌");
}
                  
try {
  const redeclareConst = "first";
  const redeclareConst = "second"; // ❌
} catch (err) {
  console.log("const cannot be redeclared ❌");
}
console.log("");
can be hoisted? ✅ ❌ ❌ 
                  
// 🟩 6. HOISTING
console.log("=== 6. Hoisting ===");
                  
console.log("var before declaration:", aVar); // ✅ undefined
var aVar = "I’m hoisted";
                  
try {
  console.log("let before declaration:", aLet); // ❌ TDZ
  let aLet = "Not hoisted";
} catch {
  console.log("let cannot be accessed before declaration ❌");
}
                  
try {
  console.log("const before declaration:", aConst); // ❌ TDZ
  const aConst = "Not hoisted";
} catch {
  console.log("const cannot be accessed before declaration ❌");
}
